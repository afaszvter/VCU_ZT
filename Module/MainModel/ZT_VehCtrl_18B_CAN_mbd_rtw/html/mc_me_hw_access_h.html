87/*!



88 * @brief MC_ME mode change.



89 *



90 * This function is changing mode in MC_ME module



91 *



92 * @param[in] base Register base address for the MC_ME instance.



93 * @param[in] mode Destination mode.



94 */



95static inline void MC_ME_ChangeMode(MC_ME_Type * base)



96{



97    uint32_t currentMode = (base->MCTL & MC_ME_MCTL_TARGET_MODE_MASK) >> MC_ME_MCTL_TARGET_MODE_SHIFT;



98



99    /* Change takes effect after mode transition. */



100    base->MCTL = MC_ME_MCTL_TARGET_MODE(currentMode) | MC_ME_MCTL_KEY(FEATURE_MC_ME_KEY);



101    base->MCTL = MC_ME_MCTL_TARGET_MODE(currentMode) | MC_ME_MCTL_KEY(FEATURE_MC_ME_KEY_INV);



102}
